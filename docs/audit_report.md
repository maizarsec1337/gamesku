### Prinsip Migrasi
- Additive only (ADD COLUMN, CREATE TABLE, ALTER INDEX)
- Non-destructive: zero DROP TABLE, TRUNCATE, DELETE
- Idempotent: mudah dijalankan ulang tanpa galat
- Backend PHP akan diperbarui setelah migrasi disetujui

### 4.5 pembayaran (payment)

```sql
CREATE TABLE `pembayaran` (
  `id`                     VARCHAR(40)  NOT NULL,
  `pesanan_id`             VARCHAR(40)  NOT NULL,
  `nominal`                BIGINT       NOT NULL DEFAULT 0 CHECK (nominal >= 0),
  `metode`                 VARCHAR(60)  NOT NULL DEFAULT '',
  `akun_pembayaran`        VARCHAR(255) NOT NULL DEFAULT '',   -- rekening/identitas sumber tanpa PIN/CVV
  `referensi`              VARCHAR(255) NULL,                  -- label transaksi
  `status`                 ENUM('Menunggu konfirmasi','Dikonfirmasi','Lunas','Gagal','Dikembalikan') NOT NULL DEFAULT 'Menunggu konfirmasi',
  `kode_idempotensi`       VARCHAR(64)  NULL,
  `transaction_id`         VARCHAR(120) NULL,
  `konfirmasi_admin`       VARCHAR(60)  NULL,
  `catatan`                VARCHAR(255) NOT NULL DEFAULT '',
  `refund_at`              DATETIME NULL DEFAULT NULL,
  `dibuat_pada`            DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP,
  `diperbaruni_pana`       DATETIME NULL DEFAULT NULL ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  UNIQUE KEY uniq_pembayaran_pesanan (pesanan_id),
  UNIQUE KEY uniq_pembayaran_transaction (transaction_id),
  KEY idx_pembayaran_status (status, dibuat_pada),
  CONSTRAINT fk_pembayaran_pesanan FOREIGN KEY (pesanan_id) REFERENCES `pesanan`(id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;


### 4.6 audit_log (riwayat aktivitas generik)

```sql
CREATE TABLE `audit_log` (
  `id`                BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  `user_id`           VARCHAR(40) NULL,
  `jenis_aktivitas`   VARCHAR(60) NOT NULL,
  `target_table`      VARCHAR(60) NULL,
  `target_id`         VARCHAR(40) NULL,
  `detail_json`       JSON NULL,
  `teks`              VARCHAR(255) NOT NULL,
  `ikon`              VARCHAR(40) NOT NULL DEFAULT 'centangLingkaran',
  `warna`             VARCHAR(20) NOT NULL DEFAULT 'hijau',
  `admin_pengguna`    VARCHAR(60) NULL,
  `dibuat_pada`       DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  KEY idx_audit_user (user_id, jenis_aktivitas),
  KEY idx_audit_jenis (jenis_aktivitas),
  KEY idx_audit_target (target_table, target_id),
  KEY idx_audit_dibuat (dibuat_pada),
  CONSTRAINT fk_audit_user FOREIGN KEY (user_id) REFERENCES `admin`(id) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;
```
